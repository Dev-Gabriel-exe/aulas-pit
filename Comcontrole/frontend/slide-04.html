fetch — a API nativa do navegador

fetch retorna uma Promise. Toda requisição de rede é assíncrona — o navegador não para enquanto espera a resposta. Por isso usamos async/await.

JavaScript — fetch GET completo
// GET simples — buscar um Pokémon
async function buscarPokemon(nome) {
  const resposta = await fetch(`https://pokeapi.co/api/v2/pokemon/${nome}`);

  // ⚠️ fetch NÃO lança erro em status 404/500 — só lança em falha de rede
  if (!resposta.ok) {
    throw new Error(`Erro ${resposta.status}: ${resposta.statusText}`);
  }

  const dados = await resposta.json(); // converte o body para objeto JS
  return dados;
}

// POST — enviar dados (padrão para chamar API de IA)
async function criarPost(titulo, corpo) {
  const resposta = await fetch('https://jsonplaceholder.typicode.com/posts', {
    method:  'POST',
    headers: { 'Content-Type': 'application/json' },
    body:    JSON.stringify({ title: titulo, body: corpo, userId: 1 }),
  });

  if (!resposta.ok) throw new Error('Falha ao criar post');
  return resposta.json();
}

// POST com Authorization header — template para APIs de IA
async function chamarIA(prompt, apiKey) {
  const res = await fetch('https://api.openai.com/v1/chat/completions', {
    method:  'POST',
    headers: {
      'Content-Type':  'application/json',
      'Authorization': `Bearer ${apiKey}`,  // ← header de autenticação
    },
    body: JSON.stringify({
      model:    'gpt-4o',
      messages: [{ role: 'user', content: prompt }],
    }),
  });
  if (!res.ok) throw new Error(`API error ${res.status}`);
  return res.json();
}
A armadilha do fetch: fetch só rejeita a Promise em caso de falha de rede (sem conexão). Um erro 404 ou 500 da API não lança exceção — você precisa checar manualmente com if (!resposta.ok). Esse é o erro mais comum de quem começa a usar APIs.
Integrando fetch com React — o padrão completo

O local certo para chamadas de API em React é dentro de um useEffect. O estado precisa de três campos: os dados, o loading e o erro.

JSX — hook useFetch reutilizável
import { useState, useEffect } from 'react';

// Hook customizado — encapsula toda lógica de fetch
export function useFetch(url) {
  const [dados,     setDados]     = useState(null);
  const [carregando, setCarregando] = useState(false);
  const [erro,       setErro]       = useState(null);

  useEffect(() => {
    if (!url) return;

    let cancelado = false;
    setCarregando(true);
    setErro(null);

    fetch(url)
      .then(res => {
        if (!res.ok) throw new Error(`Erro ${res.status}`);
        return res.json();
      })
      .then(json => { if (!cancelado) setDados(json); })
      .catch(e  => { if (!cancelado) setErro(e.message); })
      .finally(()=> { if (!cancelado) setCarregando(false); });

    return () => { cancelado = true; }; // cleanup
  }, [url]);

  return { dados, carregando, erro };
}

// ── Usar o hook ───────────────────────────────────────────────────
function ListaPosts() {
  const { dados: posts, carregando, erro } = useFetch(
    'https://jsonplaceholder.typicode.com/posts?_limit=5'
  );

  if (carregando) return <p>Carregando...</p>;
  if (erro)       return <p style={{color:'red'}}>Erro: {erro}</p>;
  if (!posts)    return null;

  return (
    <ul>
      {posts.map(p => (
        <li key={p.id}>{p.title}</li>
      ))}
    </ul>
  );
}

Tratamento de erros, estados de UI e Axios
Os 4 estados de qualquer chamada de API

Toda interface que consome API precisa lidar com exatamente esses quatro estados. Ignorar qualquer um deles resulta em UX ruim.

idle
Ainda não buscou nada. Tela inicial, aguardando ação do usuário.
loading
Requisição em andamento. Mostrar spinner ou skeleton loader.
success
Dados recebidos com sucesso. Renderizar os resultados.
error
Algo deu errado. Mostrar mensagem clara e opção de tentar novamente.
JSX — máquina de estados com useReducer
import { useReducer, useEffect } from 'react';

// Estado inicial e reducer
const INICIAL = { status: 'idle', dados: null, erro: null };

function reducer(state, action) {
  switch (action.type) {
    case 'LOADING': return { ...INICIAL, status: 'loading' };
    case 'SUCCESS': return { status: 'success', dados: action.dados, erro: null };
    case 'ERROR':   return { status: 'error',   dados: null, erro: action.erro };
    default: return state;
  }
}

function BuscaPokemon({ nome }) {
  const [state, dispatch] = useReducer(reducer, INICIAL);

  useEffect(() => {
    if (!nome) return;
    dispatch({ type: 'LOADING' });

    fetch(`https://pokeapi.co/api/v2/pokemon/${nome.toLowerCase()}`)
      .then(r => { if (!r.ok) throw new Error(`${r.status}`); return r.json(); })
      .then(d  => dispatch({ type: 'SUCCESS', dados: d }))
      .catch(e  => dispatch({ type: 'ERROR',   erro: e.message }));
  }, [nome]);

  const { status, dados, erro } = state;

  return (
    <div>
      {status === 'idle'    && <p>Digite um nome para buscar.</p>}
      {status === 'loading' && <div className="spinner">Carregando...</div>}
      {status === 'error'   && (
        <div>
          <p>Erro: {erro}</p>
          <button onClick={() => dispatch({ type: 'LOADING' })}>Tentar novamente</button>
        </div>
      )}
      {status === 'success' && <p>{dados.name} — #{dados.id}</p>}
    </div>
  );
}
Axios — quando o fetch nativo não basta

Axios é um cliente HTTP que resolve as principais dores do fetch: rejeita automaticamente em erros HTTP, intercepta requisições e respostas, e tem cancelamento nativo com AbortController.

fetch — verboso

fetch
const res = await fetch(url, {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': `Bearer ${key}`
  },
  body: JSON.stringify(dados)
});
if (!res.ok) throw new Error(...);
const json = await res.json();

Axios — enxuto

Axios
const { data } = await axios.post(url, dados, {
  headers: {
    'Authorization': `Bearer ${key}`
  }
});
// erros HTTP já lançam exception
// data já é o objeto JS direto
JavaScript — Axios com instância configurada
import axios from 'axios';

// Instância configurada — padrão para projetos reais
const api = axios.create({
  baseURL: 'https://api.openai.com/v1',
  headers: {
    'Authorization': `Bearer ${import.meta.env.VITE_OPENAI_KEY}`,
    'Content-Type':  'application/json',
  },
  timeout: 30000, // 30 segundos
});

// Interceptor — tratamento global de erros
api.interceptors.response.use(
  response => response,
  error => {
    if (error.response?.status === 401) {
      alert('Chave de API inválida. Verifique suas configurações.');
    }
    if (error.response?.status === 429) {
      alert('Limite de requisições atingido. Aguarde um momento.');
    }
    return Promise.reject(error);
  }
);

// Chamar a API de chat
async function chat(mensagens) {
  const { data } = await api.post('/chat/completions', {
    model:    'gpt-4o',
    messages: mensagens,
  });
  return data.choices[0].message.content;
}

O problema — chave de API no código

Esse é o erro mais perigoso que você pode cometer ao trabalhar com APIs de IA. Uma chave da OpenAI exposta no GitHub pode gerar centenas de dólares de cobrança em minutos.

❌ NUNCA faça isso

JSX — PERIGOSO

// ↑ vai para o git, vai para o GitHub,
//   vai ser varrido por bots em minutos

const res = await fetch(url, {
  headers: {
    'Authorization': `Bearer ${API_KEY}`
  }
});

✅ Faça assim

JSX — seguro com .env
// app.jsx — correto
const API_KEY = import.meta.env.VITE_OPENAI_KEY;
// ↑ lê do .env — nunca vai para o git

const res = await fetch(url, {
  headers: {
    'Authorization': `Bearer ${API_KEY}`
  }
});
Bots do GitHub escaneiam commits em tempo real procurando por chaves de API. Uma chave da OpenAI exposta por menos de 1 minuto já foi suficiente para gerar cobranças de $500+. Não existe "só por um momento" — use .env desde o primeiro commit.

Juntando tudo — hook useIA pronto para a Semana 5

Este é o hook que você vai usar na próxima semana com a API real. Já está estruturado com .env, Axios, estados e tratamento de erros.

JSX — hooks/useIA.js — template Semana 5
import { useState } from 'react';
import axios from 'axios';

// Instância Axios — chave vem do .env
const apiClient = axios.create({
  baseURL: 'https://api.openai.com/v1',
  headers: {
    'Authorization': `Bearer ${import.meta.env.VITE_OPENAI_KEY}`,
    'Content-Type':  'application/json',
  },
});

export function useIA() {
  const [resposta,    setResposta]    = useState('');
  const [carregando,  setCarregando]  = useState(false);
  const [erro,        setErro]        = useState(null);

  const perguntar = async (prompt, historico = []) => {
    setCarregando(true);
    setErro(null);
    setResposta('');

    try {
      const { data } = await apiClient.post('/chat/completions', {
        model:    'gpt-4o',
        messages: [
          ...historico,
          { role: 'user', content: prompt }
        ],
      });
      const texto = data.choices[0].message.content;
      setResposta(texto);
      return texto;

    } catch (e) {
      const msg = e.response?.data?.error?.message || e.message;
      setErro(msg);
      throw e;
    } finally {
      setCarregando(false);
    }
  };

  return { resposta, carregando, erro, perguntar };
}

// Semana 5: substitua apiClient pelo modelo real e pronto!
Semana 5 começa aqui. O hook useIA acima é a base do que você vai construir na próxima aula. Só falta colocar a chave real no .env e o chatbot da Sem. 2 vai começar a responder de verdade.
